# ============================================================================
# LogoDet-3K setup 阶段依赖锁定
#
# 锁定策略（两档，故意区别对待）：
#
#   [精确锁] 会改变指标数字的包 —— 版本一变，AP 可能就变。
#            这些必须 == 精确版本，S0-G1 门会逐项对账。
#
#   [范围锁] 纯工具链 —— 只影响使用体验，不影响任何上报数字。
#            用兼容范围，避免为了一个 CLI 补丁号去打断整个环境。
#
# 平台：macOS 14+ / Apple Silicon (arm64) / Python 3.11
#   为什么是 3.11 而不是 3.13：torch、torchvision、pycocotools 的
#   arm64 wheel 在 3.11 上最完整；3.13 容易掉进源码编译。
# ============================================================================

# ---- [精确锁] 数值相关 ------------------------------------------------------

# torch / torchvision 必须成对，版本错配会在 import 时炸 C++ ABI
torch==2.5.1
torchvision==0.20.1

# 关键：不用 numpy 2.x。
#   本机 conda base 是 numpy 2.4.4，但 pycocotools 与一批老库对
#   numpy 2 的 ABI 变更兼容性不稳；1.26.4 是 1.x 的最后稳定版。
#   S0-G3 的玩具 COCO 冒烟（AP 必须 == 1.0）就是专门用来抓这个的。
numpy==1.26.4

# COCO 官方参考评测实现。2.0.8 有 macosx arm64 预编译 wheel，免编译。
pycocotools==2.0.8

# 图像解码。EXIF 旋转的处理行为跨版本有差异 → 会影响 W/H → 影响 area_bin
# → 影响尺寸切片，所以必须锁死。
pillow==10.4.0

# 中间表读写
pandas==2.2.3
pyarrow==17.0.0

# OWLv2。post_process_grounded_object_detection 的签名与 padded-square
# 语义在 4.4x 之间有过调整，锁死避免坐标口径漂移。
transformers==4.46.3

# XML 解析。recover=True 的容错行为跨版本有差异 → 影响脏数据处置 → 锁死。
lxml==5.3.0

# ---- [范围锁] 工具链 --------------------------------------------------------

pyyaml>=6.0,<7
tqdm>=4.66,<5
matplotlib>=3.9,<4

# transformers 的 Owlv2ImageProcessor 的**隐式**依赖：它用 scipy 做
# anti-aliasing resize，缺了会在 S7b 第一次构造 processor 时抛
# `Owlv2ImageProcessor requires scipy`。
#
# ⚠️ 这一行是补进来的 —— 早期锁文件漏了它，导致「全新环境按 s0_setup_env.sh
#    建好后能跑通 S0–S7，却在 S7b 挂掉」。范围锁而非精确锁，因为它只参与
#    图像预处理的插值实现，不参与任何指标计算。
#
# 本机若因文件守卫拦住源码构建而装不上，改用只取 wheel：
#     pip install --only-binary :all: scipy
scipy>=1.11,<2

# 单测
pytest>=8.3,<9

# ---- 已移除：kaggle CLI ----------------------------------------------------
#
# 原本这里有 kaggle>=1.6.14,<1.7，已去掉，两个理由：
#
#   1. 它是本锁文件里唯一需要源码构建的包，而本机的文件守卫会拦截
#      pip 构建时的 mkdir（返回 Node 风格的 EEXIST），导致**整个 pip 事务
#      失败**——其余 12 个包一个都装不上。为一个下载工具赔上整个环境不值。
#
#   2. 3.27GB 的下载本来就更适合 curl：
#        curl -L -C - -u "$USER:$KEY" \
#          -o LogoDet-3K.zip \
#          https://www.kaggle.com/api/v1/datasets/download/lyly99/logodet3k
#      -C - 提供断点续传，CLI 反而没有可靠的续传保证。
#      凭据仍从 ~/.kaggle/kaggle.json 读，认证方式完全一致。
#
# 具体实现见 scripts/s1_download.py。
